// 重新精簡並嚴謹化篩選邏輯 const loadData = async () => { try { // 1. 抓取表單回應資料 const resForm = await fetch(formUrl); const textForm = await resForm.text(); const jsonForm = JSON.parse(textForm.substring(textForm.indexOf('{'), textForm.lastIndexOf('}') + 1)); const rawFormRows = jsonForm.table.rows || []; // 2. 讀取輔具清單主表 const resList = await fetch(listUrl); const textList = await resList.text(); const jsonList = JSON.parse(textList.substring(textList.indexOf('{'), textList.lastIndexOf('}') + 1)); const rows = jsonList.table.rows; const cols = jsonList.table.cols; equipments.value = rows.map(row => { let item = {}; row.c.forEach((cell, i) => { const label = cols[i].label.trim(); item[label] = cell ? String(cell.v).trim() : ''; }); const name = item['輔具名稱']; // 確保這裡與試算表欄位名稱完全一致 const total = parseInt(item['數量']) || 0; // 嚴格計算：只計算「借用輔具名稱」這一欄內容「完全等於」該輔具名稱的次數 let rentedCount = 0; rawFormRows.forEach(fRow => { // 假設您的表單中「借用輔具名稱」是第 5 欄 (索引 4，請依實際情況調整) const rentedName = fRow.c[4] ? String(fRow.c[4].v).trim() : ''; if (rentedName === name) { rentedCount++; } }); return { name: name, total: total, remaining: Math.max(0, total - rentedCount), // ... 其他屬性 }; }); } catch (e) { console.error("載入失敗", e); } };
